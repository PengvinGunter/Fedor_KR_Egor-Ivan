create table roles(
role_id int primary key identity(1,1),
role_name nvarchar(50)
);
create table auto_brand(
brand_id int primary key identity(1,1),
brand_name nvarchar(40)
);
create table auto(
auto_id int primary key identity(1,1),
VIN nvarchar(17) not null unique,
year_release int not null,
brand_id int not null,
foreign key(brand_id) references auto_brand(brand_id),
mileage int
)
create table hours_work(
day_id int primary key identity(1,1),
hours_open time,
hours_close time
)
create table users(
user_id int primary key identity(1,1),
role_id int,
foreign key(role_id) references roles(role_id),
user_name nvarchar(40),
user_secondname nvarchar(40),
phone nvarchar(10) not null unique,
)
create table rental_point(
point_id int primary key identity(1,1),
city nvarchar(30) not null,
adress nvarchar(100) not null,
hours_work int not null,
foreign key(hours_work) references hours_work(day_id),
capacity_park int,
manager_id int,
foreign key (manager_id) references users(user_id)
)
create table contract_status(
status_id int primary key identity(1,1),
status_name nvarchar(50),
)
create table contract(
contract_id int primary key identity(1,1),
point_id int,
foreign key (point_id) references rental_point(point_id),
start_day datetime not null,
start_end datetime,
auto_id int,
foreign key(auto_id) references auto(auto_id),
status_id int,
foreign key(status_id) references contract_status(status_id)
)
create table user_booking(
booking_id int primary key identity(1,1),
client_id int,
foreign key (client_id) references users(user_id),
contract_id int,
foreign key (contract_id) references contract(contract_id)
)
